    function renderMarksheet() {

      const el = document.getElementById('marksheetList');

      const isTeacher = activeRole === 'Teacher';

      const batch = getTargetBatch();

      let stList = db.students;

      if (!isTeacher) {

        stList = db.students.filter(s => s.gender === batch);

      }

      el.innerHTML = stList.map(s => {

        const tests = db.tests.filter(t => t.studentId === s.id);

        const got = tests.reduce((a,b)=>a+Number(b.marks),0);

        const tot = tests.reduce((a,b)=>a+Number(b.total),0);

        const pct = tot > 0 ? Math.round((got/tot)*100) : 0;

        return `

          <div class="ios-glass-card p-4 space-y-2 text-xs">

            <div class="flex justify-between items-center">

              <b class="text-stone-900 text-sm font-black">${s.name}</b>

              <span class="text-amber-800 font-bold bg-amber-100 px-2.5 py-0.5 rounded-full border border-amber-300">${s.std} • ${s.gender}</span>

            </div>

            <p class="text-stone-500 text-[11px]">Total Tests: <b>${tests.length}</b></p>

            <div class="flex justify-between border-t border-amber-900/10 pt-2 font-bold">

              <span class="text-stone-500">Cumulative Score:</span>

              <span class="${pct>=40?'text-emerald-600':'text-rose-600'} text-sm font-black">${pct}% (${got}/${tot})</span>

            </div>

          </div>

        `;

      }).join('');

    }
